import { Card, CardGrid } from '@astrojs/starlight/components';

```bash
npm i -g cloudhop
cloudhop setup
hop acme-prod s3 ls
```

<CardGrid stagger>
  <Card title="One login per portal" icon="rocket">
    Log in once to each IAM Identity Center portal. cloudhop discovers every
    account and role you have access to.
  </Card>
  <Card title="Logs in only when needed" icon="approve-check">
    Sessions are checked when you use an alias. Expired? cloudhop renews it
    silently or opens the browser, then runs your command.
  </Card>
  <Card title="No access keys" icon="lock">
    Only temporary credentials: AWS SSO for companies, `aws login` for
    personal accounts.
  </Card>
  <Card title="Works with your tools" icon="puzzle">
    Profiles live in `~/.aws/config`, so Terraform, SDKs and IDEs use the
    same aliases.
  </Card>
</CardGrid>
